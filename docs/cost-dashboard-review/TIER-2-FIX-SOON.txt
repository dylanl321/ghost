    ```typescript
    // app/utils/chart-options/tooltip-factory.ts

    import { formatCurrency } from '~/utils/formatters'

    export interface TooltipField {
      label: string
      value: number | string
      format?: 'currency' | 'percent' | 'number' | 'text'
      color?: string
      bold?: boolean
    }

    export interface TooltipConfig {
      title?: string | ((params: any) => string)
      fields: TooltipField[] | ((params: any, data: any) => TooltipField[])
      showDot?: boolean         // Show series color dot
      showDelta?: boolean       // Show change between values
      deltaFields?: [string, string]  // Field names to calculate delta
      currencyAbbreviated?: boolean
      footer?: string | ((params: any) => string)
    }

    export function createTooltipFormatter(config: TooltipConfig) {
      return function(params: any) {
        const p = Array.isArray(params) ? params[0] : params
        const dataIndex = p.dataIndex ?? 0

        let html = '<div class="chart-tooltip">'

        // Title
        const title = typeof config.title === 'function'
          ? config.title(p) : (config.title || p.name || '')
        if (title) {
          html += `<div class="font-semibold mb-1">${title}</div>`
        }

        // Fields
        const fields = typeof config.fields === 'function'
          ? config.fields(p, p.data) : config.fields

        for (const field of fields) {
          const formatted = formatFieldValue(field)
          const colorDot = config.showDot && field.color
            ? `<span style="display:inline-block;width:8px;height:8px;border-radius:50%;background:${field.color};margin-right:4px"></span>`
            : ''
          const weight = field.bold ? 'font-semibold' : ''
          html += `<div class="${weight}">${colorDot}${field.label}: ${formatted}</div>`
        }

        // Delta
        if (config.showDelta && config.deltaFields) {
          const [fieldA, fieldB] = config.deltaFields
          const valA = fields.find(f => f.label === fieldA)?.value as number
          const valB = fields.find(f => f.label === fieldB)?.value as number
          if (valA != null && valB != null) {
            const delta = valA - valB
            const color = delta >= 0 ? '#ef4444' : '#22c55e'
            html += `<div style="color:${color}">Delta: ${formatCurrency(Math.abs(delta))}</div>`
          }
        }

        // Footer
        if (config.footer) {
          const footerText = typeof config.footer === 'function'
            ? config.footer(p) : config.footer
          html += `<div class="text-xs text-gray-400 mt-1">${footerText}</div>`
        }

        html += '</div>'
        return html
      }
    }

    function formatFieldValue(field: TooltipField): string {
      const val = field.value
      switch (field.format) {
        case 'currency': return formatCurrency(val as number, 2)
        case 'percent': return `${(val as number).toFixed(1)}%`
        case 'number': return (val as number).toLocaleString()
        default: return String(val)
      }
    }
    ```

  Step 1: Extract shared trend comparison builder
    ```typescript
    // In useChartRegistry.ts or new file app/utils/chart-extractors/trend-comparison.ts

    interface TrendComparisonConfig {
      groupField: 'linesOfBusiness' | 'departmentNumber'
      groupSubField: 'lineOfBusiness' | 'departmentId'  // field within group
      labelField: string          // display name field
      defaultLabel: string        // e.g., 'Unknown LOB' or 'Unknown Dept'
      minTotalThreshold: number   // minimum total to include group
      topLimit: number            // max groups to show
      chartTitle: string
    }

    function buildTrendComparison(
      dataset: NormalizedDataset,
      config: TrendComparisonConfig
    ): ChartData | null {
      const accounts = dataset.accountsList || []
      const accountGroupMap = buildAccountToGroupMap(
        accounts, config.groupField, config.groupSubField
      )

      const dataSource = dataset.mode === 'proforma'
        ? dataset.historicalOrgWide?.proforma
        : dataset.historicalOrgWide
      const resultsByTime = dataSource?.resultsByTime || []
      if (!resultsByTime.length) return null

      const seriesMap = new Map<string, number[]>()
      const dates: string[] = []

      resultsByTime.forEach((bucket: any) => {
        const dateStr = bucket.TimePeriod?.Start
        if (!dateStr) return
        dates.push(dateStr)

        const groups = bucket.Groups || []
        const dateTotals = new Map<string, number>()

        groups.forEach((group: any) => {
          const accountId = group.Keys?.[0]
          const amount = parseFloat(group.Metrics?.UnblendedCost?.Amount || '0')
          const groupName = accountGroupMap.get(accountId) || config.defaultLabel

          dateTotals.set(groupName, (dateTotals.get(groupName) || 0) + amount)
        })

        for (const [name, total] of dateTotals) {
          if (!seriesMap.has(name)) {
            seriesMap.set(name, new Array(dates.length - 1).fill(0))
          }
          seriesMap.get(name)!.push(total)
        }

        // Pad series that didn't appear in this bucket
        for (const [name, values] of seriesMap) {
          if (values.length < dates.length) {
            values.push(0)
          }
        }
      })

      // Filter by threshold, sort by total, limit to top N
      const ranked = [...seriesMap.entries()]
        .map(([name, values]) => ({
          name,
          values,
          total: values.reduce((s, v) => s + v, 0)
        }))
        .filter(s => s.total >= config.minTotalThreshold)
        .sort((a, b) => b.total - a.total)
        .slice(0, config.topLimit)

      return {
        dates: dates.map(d => formatDate(d)),
        series: ranked.map(s => ({ name: s.name, data: s.values })),
        title: config.chartTitle
      }
    }
    ```

7. NORMALIZE API RESPONSE SHAPES - Consistent Server Responses
================================================================================

PROBLEM:
  Cost API endpoints return different nesting levels and structures, requiring
  complex unwrapping logic on the client. The inconsistency makes the code
  brittle and hard to maintain.

  Current response shape variations:
    - Summary enhanced: { accounts[], summary{} }
    - Summary raw: { resultsByTime[] } or { proforma.rawData.resultsByTime }
    - Services: { proforma.resultsByTime } (no rawData wrapper!)
    - Analysis: { budgetAnalysis.byAccount[] }
    - Historical: { resultsByTime[] } or { proforma.rawData.resultsByTime }

  This forces cost-data-unwrap.ts to have multiple code paths:
    - unwrapCostData() handles top-level wrapper
    - getResultsByTime() handles proforma nesting variations (lines 50-79)
    - Multiple fallback chains: proforma.rawData.resultsByTime || proforma.resultsByTime

AFFECTED FILES & SPECIFIC LOCATIONS:

  Server endpoints:
    - server/api/costs/summary.get.ts: Returns different shape in enhanced vs basic mode
    - server/api/costs/historical.get.ts: Returns resultsByTime or proforma-wrapped
    - server/api/costs/forecast.get.ts: Own response structure
    - server/api/costs/analysis.get.ts: budgetAnalysis wrapper

  Client unwrapping:
    - app/utils/cost-data-unwrap.ts: All unwrap functions
      - Lines 50-79: getResultsByTime() with dual proforma path handling
      - Tries proforma.rawData.resultsByTime, then proforma.resultsByTime
    - app/composables/useCostData.ts: Calls unwrap utilities throughout

WHAT TO DO:

  Step 1: Define a canonical response envelope

8. IMPROVE ERROR HANDLING - Categorize, Retry, Distinguish States
================================================================================

PROBLEM:
  Multiple issues across the error handling pipeline:

  A) useCostData.ts catch block (lines 328-337) treats all errors identically.
     Network errors, timeouts, and cancellations all become the same generic
     error message. No retry logic for transient failures.

  B) useAccountCosts.ts has silent catch blocks that return empty arrays without
     user feedback (e.g., fetchLinesOfBusiness lines 213-217).

  C) ChartsRenderer.vue doesn't distinguish loading vs no-data vs error states.
     All three show the same "No data" message to users.

  D) API error messages are not user-friendly. Validation errors return technical
     strings that users can't act on.

  app/composables/useCostData.ts

    - Line 218: if (!chart.extractor) return false - silently skips charts

  Server endpoints:
    - server/api/costs/summary.get.ts line 249: Technical error messages

WHAT TO DO:

  Step 1: Categorize errors in useCostData.ts
    ```typescript
    catch (e: any) {
      // Expected: User changed filters, previous request cancelled
      if (e?.name === 'AbortError') {
        isDeferredLoading.value = false
        return
      }

      // Expected: Stale response from previous generation
      if (thisGeneration !== fetchGeneration) {
        isDeferredLoading.value = false
        return
      }

      // Transient: Network or timeout errors - retry
      if (isTransientError(e)) {
        if (retryCount < MAX_RETRIES) {
          retryCount++
          console.warn(`Cost fetch retry ${retryCount}/${MAX_RETRIES}:`, e.message)
          setTimeout(() => fetchCosts(), RETRY_DELAY * retryCount)
          return
        }
      }

      // Actual error: Surface to user
      console.error('Cost fetch failed:', { error: e, query: query.value })
      fetchError.value = getUserFriendlyError(e)
      isFetching.value = false
      isDeferredLoading.value = false
    }

    function isTransientError(e: any): boolean {
      return e?.status === 429          // Rate limited
        || e?.status === 503            // Service unavailable
        || e?.code === 'ECONNRESET'     // Connection reset
        || e?.message?.includes('timeout')
    }

    function getUserFriendlyError(e: any): string {
      if (e?.status === 403) return 'You do not have permission to view cost data for this account.'
      if (e?.status === 429) return 'Too many requests. Please wait a moment and try again.'
      if (e?.status >= 500) return 'The cost service is temporarily unavailable. Please try again.'
      return e?.message || 'Failed to load cost data. Please try refreshing.'
    }
    ```

    ```typescript
      status: 'loading' | 'success' | 'noData' | 'error'

    function getChartState(chart: ChartDefinition, dataset: NormalizedDataset | null): ChartState {
      if (!dataset) return { status: 'loading' }
      try {

        if (!data || (Array.isArray(data) && data.length === 0)) {
          return { status: 'noData' }

      }
    ```
